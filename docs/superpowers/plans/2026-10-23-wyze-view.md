# Wyze View Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a GTK3 desktop app that shows live RTSP feeds from Wyze cameras in a click-to-enlarge grid, with per-camera audio, snapshots, and fullscreen.

**Architecture:** One Python process. Each camera is an `mpv` subprocess embedded into a `Gtk.EventBox` via `--wid` on X11. mpv handles RTSP, hardware decode, audio, screenshots, and reconnects; the Python app owns config, layout, and UI state.

**Tech Stack:** Python 3.13, PyGObject/GTK3 (installed), mpv 0.40 (apt), pytest (dev-only, in a venv), TOML config via `tomllib`.

**Spec:** `docs/superpowers/specs/2026-10-23-wyze-view-design.md`

## Global Constraints

- Target: Debian 13 (trixie), XFCE, **X11** session; X11 is required for `--wid` embedding.
- Runtime dependency: `mpv` only. Do not add other runtime dependencies.
- Dev dependency: `pytest`, installed only into `.venv/`.
- Python `>=3.11` (uses `tomllib`).
- Grid uses `sub_url` when configured; enlarge uses `url`.
- Audio: all muted at start; exactly one unmuted camera when enlarged; none in grid.
- Config lives at `~/.config/wyze-view/cameras.toml`, directory `0700`, file `0600`.
- Never print or log RTSP passwords; redact userinfo in all output.
- No recording, motion detection, PTZ, auto-discovery, remote access, or Roku support.
- App binds no network ports.

```python
from __future__ import annotations

import sys
import tomllib
from dataclasses import dataclass
from pathlib import Path

DEFAULT_CONFIG = Path.home() / ".config" / "wyze-view" / "cameras.toml"

DEFAULT_TOML = """\
# Wyze View cameras. Add one [[camera]] block per feed.
# url is required; sub_url is optional and used for the CPU-cheap grid view.
#
# [[camera]]
# name = "Corner"
# url = "rtsp://user:pass@192.168.1.165:554/stream0/"
# sub_url = "rtsp://user:pass@192.168.1.165:554/stream1/"
"""


@dataclass(frozen=True)
class Camera:
    name: str
    url: str
    sub_url: str | None = None


def _warn(message: str) -> None:
    print(f"wyze-view: {message}", file=sys.stderr)


def _create_default(path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
    with path.open("x") as fh:
        fh.write(DEFAULT_TOML)
    path.chmod(0o600)


def load_cameras(path: Path = DEFAULT_CONFIG) -> list[Camera]:
    if not path.exists():
        _create_default(path)
        return []

    raw = tomllib.loads(path.read_text())
    entries = raw.get("camera", [])
    if not isinstance(entries, list):
        _warn(f"{path}: 'camera' must be an array of tables")
        return []

    cameras: list[Camera] = []
    seen: set[str] = set()
    for index, entry in enumerate(entries, 1):
        if not isinstance(entry, dict):
            _warn(f"camera #{index}: not a table, skipped")
            continue

        name = entry.get("name")
        url = entry.get("url")
        sub_url = entry.get("sub_url")

        if not isinstance(name, str) or not name.strip():
            _warn(f"camera #{index}: missing name, skipped")
            continue
        if not isinstance(url, str) or not url.strip():
            _warn(f"camera '{name}': missing url, skipped")
            continue
        if not isinstance(sub_url, str) or not sub_url.strip():
            sub_url = None
        if name in seen:
            _warn(f"camera '{name}': duplicate name, skipped")
            continue

        seen.add(name)
        cameras.append(
            Camera(
                name=name.strip(),
                url=url.strip(),
                sub_url=sub_url.strip() if sub_url else None,
            )
        )

    return cameras
```

```python
from __future__ import annotations

import re
from datetime import datetime
from pathlib import Path


def sanitize_name(name: str) -> str:
    cleaned = re.sub(r"[^A-Za-z0-9._-]+", "_", name).strip("._-")
    return cleaned or "camera"


def snapshot_path(directory: Path, name: str, when: datetime) -> Path:
    return Path(directory) / f"{sanitize_name(name)}-{when:%Y%m%d-%H%M%S}.jpg"


def redact_url(url: str) -> str:
    return re.sub(r"//[^/@]*@", "//***@", url)
```

```python
from __future__ import annotations

import math


def grid_dims(n: int) -> tuple[int, int]:
    if n <= 0:
        return (0, 0)
    if n == 1:
        return (1, 1)
    if n == 2:
        return (1, 2)
    if n <= 4:
        return (2, 2)
    if n <= 9:
        return (3, 3)
    if n <= 16:
        return (4, 4)
    cols = math.ceil(math.sqrt(n))
    rows = math.ceil(n / cols)
    return (rows, cols)


def grid_position(index: int, rows: int, cols: int) -> tuple[int, int]:
    if cols <= 0:
        return (0, 0)
    return (index // cols, index % cols)
```

```python
from __future__ import annotations

import json
import socket
import subprocess
import threading
import time
from pathlib import Path
from typing import Callable

from .config import Camera


def build_mpv_args(xid: int, socket_path: Path, url: str) -> list[str]:
    return [
        "mpv",
        f"--wid={xid}",
        "--no-config",
        "--no-border",
        "--no-osc",
        "--osd-level=0",
        f"--input-ipc-server={socket_path}",
        "--rtsp-transport=tcp",
        "--hwdec=auto-safe",
        "--mute=yes",
        "--screenshot-format=jpg",
        "--really-quiet",
        "--idle=yes",
        "--force-window=yes",
        url,
    ]


def backoff_delay(attempt: int) -> float:
    return float(min(2 ** (attempt + 1), 30))


def ipc_command(socket_path: Path, command: list, timeout: float = 3.0) -> dict:
    deadline = time.monotonic() + timeout
    while not socket_path.exists():
        if time.monotonic() > deadline:
            raise TimeoutError(f"mpv IPC socket did not appear: {socket_path}")
        time.sleep(0.05)

    with socket.socket(socket.AF_UNIX, socket.SOCK_STREAM) as sock:
        sock.settimeout(timeout)
        sock.connect(str(socket_path))
        payload = json.dumps({"command": command, "request_id": 1}) + "\n"
        sock.sendall(payload.encode())
        buffer = b""
        while b"\n" not in buffer:
            chunk = sock.recv(4096)
            if not chunk:
                break
            buffer += chunk
        line = buffer.split(b"\n", 1)[0]
        return json.loads(line or b"{}")


class Player:
    def __init__(
        self,
        camera: Camera,
        xid: int,
        socket_path: Path,
        on_state: Callable[[str], None] | None = None,
    ) -> None:
        self.camera = camera
        self.xid = xid
        self.socket_path = Path(socket_path)
        self.on_state = on_state or (lambda _state: None)
        self._url = camera.sub_url or camera.url
        self._proc: subprocess.Popen | None = None
        self._stopping = False
        self._attempt = 0
        self._state = "stopped"

    @property
    def state(self) -> str:
        return self._state

    def _set_state(self, state: str) -> None:
        self._state = state
        self.on_state(state)

    def start(self) -> None:
        self.socket_path.unlink(missing_ok=True)
        self._spawn()
        threading.Thread(target=self._watch, daemon=True).start()

    def _spawn(self) -> None:
        self._proc = subprocess.Popen(
            build_mpv_args(self.xid, self.socket_path, self._url),
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
        )
        self._set_state("playing")

    def _watch(self) -> None:
        started = time.monotonic()
        while not self._stopping:
            if self._proc is None:
                time.sleep(0.1)
                continue
            self._proc.wait()
            if self._stopping:
                break
            if time.monotonic() - started > 30:
                self._attempt = 0
            self._set_state("offline")
            time.sleep(backoff_delay(self._attempt))
            if self._stopping:
                break
            self._attempt += 1
            try:
                self.socket_path.unlink(missing_ok=True)
                self._spawn()
                started = time.monotonic()
            except OSError:
                self._set_state("offline")

    def load(self, url: str) -> None:
        self._url = url
        if self._proc and self._proc.poll() is None:
            try:
                ipc_command(self.socket_path, ["loadfile", url, "replace"])
            except (OSError, TimeoutError):
                pass

    def set_muted(self, muted: bool) -> None:
        if self._proc and self._proc.poll() is None:
            try:
                ipc_command(self.socket_path, ["set_property", "mute", muted])
            except (OSError, TimeoutError):
                pass

    def set_paused(self, paused: bool) -> None:
        if self._proc and self._proc.poll() is None:
            try:
                ipc_command(self.socket_path, ["set_property", "pause", paused])
            except (OSError, TimeoutError):
                pass

    def snapshot(self, path: Path) -> None:
        ipc_command(self.socket_path, ["screenshot-to-file", str(path), "video"])

    def stop(self) -> None:
        self._stopping = True
        if self._proc and self._proc.poll() is None:
            try:
                ipc_command(self.socket_path, ["quit"], timeout=1)
            except (OSError, TimeoutError):
                pass
            try:
                self._proc.wait(timeout=2)
            except subprocess.TimeoutExpired:
                self._proc.terminate()
        self.socket_path.unlink(missing_ok=True)
        self._set_state("stopped")
```

```python
            ("Mute", self.toggle_mute),
            ("Fullscreen", self.toggle_fullscreen),
        ):
            button = Gtk.Button(label=label)
            button.connect("clicked", lambda _b, cb=callback: cb())
            header.pack_start(button)

        self.status = Gtk.Label(label="")
        self.status.set_xalign(0.0)
        self.status.get_style_context().add_class("status")

        self.fixed = Gtk.Fixed()
        self._last_alloc = (0, 0)

        box = Gtk.Box(orientation=Gtk.Orientation.VERTICAL)
        box.pack_start(self.fixed, True, True, 0)
        box.pack_start(self.status, False, False, 0)
        self.add(box)

        provider = Gtk.CssProvider()
        provider.load_from_data(CSS)
        Gtk.StyleContext.add_provider_for_screen(
            Gdk.Screen.get_default(),
            provider,
            Gtk.STYLE_PROVIDER_PRIORITY_APPLICATION,
        )

        self.fixed.connect("size-allocate", self._on_fixed_allocate)
        self.connect("key-press-event", self._on_key)
        self.connect("destroy", Gtk.main_quit)

        self.reload()

    def _set_status(self, text: str) -> None:
        self.status.set_text(text)

    def reload(self) -> None:
        for player in self.players.values():
            player.stop()
        self.players.clear()
        self.tiles.clear()
        self.enlarged = None
        self.enlarged_muted = False
        for child in self.fixed.get_children():
            self.fixed.remove(child)

        self.cameras = load_cameras(self.config_path)
        if not self.cameras:
            self._set_status(f"No cameras configured in {self.config_path}")
            return

        for camera in self.cameras:
            tile = Gtk.EventBox()
            tile.get_style_context().add_class("tile")
            tile.add_events(Gdk.EventMask.BUTTON_PRESS_MASK)
            tile.connect("button-press-event", self._on_tile_click, camera.name)
            tile.set_size_request(320, 180)
            self.fixed.put(tile, 0, 0)
            self.tiles[camera.name] = tile

        self._layout()
        self.show_all()
        GLib.idle_add(self._start_players)
        self._set_status(f"{len(self.cameras)} cameras")

    def _on_fixed_allocate(self, _widget, allocation) -> bool:
        size = (allocation.width, allocation.height)
        if size != self._last_alloc:
            self._last_alloc = size
            self._layout()
        return False

    def _layout(self) -> None:
        if not self.cameras:
            return
        width = max(self._last_alloc[0], 1)
        height = max(self._last_alloc[1], 1)

        if self.enlarged:
            for name, tile in self.tiles.items():
                tile.set_visible(name == self.enlarged)
            tile = self.tiles[self.enlarged]
            tile.set_size_request(width, height)
            self.fixed.move(tile, 0, 0)
            return

        rows, cols = grid_dims(len(self.cameras))
        cell_w = max(width // max(cols, 1), 1)
        cell_h = max(height // max(rows, 1), 1)
        for index, camera in enumerate(self.cameras):
            row, col = grid_position(index, rows, cols)
            tile = self.tiles[camera.name]
            tile.set_visible(True)
            tile.set_size_request(cell_w, cell_h)
            self.fixed.move(tile, col * cell_w, row * cell_h)

    def _start_players(self) -> bool:
        pending = [c for c in self.cameras if c.name not in self.players]
        if not pending:
            return False
        windows = [self.tiles[c.name].get_window() for c in pending]
        if any(window is None for window in windows):
            GLib.timeout_add(100, self._start_players)
            return False

        for camera, window in zip(pending, windows):
            socket_path = Path(self._tmp.name) / f"{camera.name}.sock"
            player = Player(
                camera,
                window.get_xid(),
                socket_path,
                on_state=lambda state, name=camera.name: GLib.idle_add(
                    self._on_state, name, state
                ),
            )
            player.start()
            self.players[camera.name] = player
        return False

    def _on_state(self, name: str, state: str) -> None:
        if state == "offline":
            self._set_status(f"{name}: offline — retrying")

    def _on_tile_click(self, _widget, event, name: str) -> bool:
        if event.button != 1:
            return False
        if self.enlarged == name:
            self._collapse()
        else:
            self._enlarge(name)
        return True

    def _enlarge(self, name: str) -> None:
        self.enlarged = name
        camera = next(c for c in self.cameras if c.name == name)
        for other_name, other in self.players.items():
            if other_name != name:
                other.set_muted(True)
                other.set_paused(True)
        player = self.players.get(name)
        if player:
            player.load(camera.url)
            player.set_paused(False)
            player.set_muted(False)
        self.enlarged_muted = False
        self._layout()
        self._set_status(f"Viewing {name}")

    def _collapse(self) -> None:
        name = self.enlarged
        self.enlarged = None
        camera = next((c for c in self.cameras if c.name == name), None)
        player = self.players.get(name) if name else None
        if camera and player:
            player.load(camera.sub_url or camera.url)
        for other in self.players.values():
            other.set_muted(True)
            other.set_paused(False)
        self._layout()
        self._set_status(f"{len(self.cameras)} cameras")

    def snapshot(self) -> None:
        name = self.enlarged or (self.cameras[0].name if self.cameras else None)
        if not name:
            self._set_status("No camera to snapshot")
            return
        player = self.players.get(name)
        if not player:
            self._set_status("Camera not ready")
            return
        directory = Path.home() / "Pictures" / "cameras"
        directory.mkdir(parents=True, exist_ok=True)
        path = snapshot_path(directory, name, datetime.now())
        try:
            player.snapshot(path)
            self._set_status(f"Saved {path}")
        except Exception as exc:  # noqa: BLE001 - show any failure in the UI
            self._set_status(f"Snapshot failed: {exc}")

    def toggle_mute(self) -> None:
        if not self.enlarged:
            self._set_status("Enlarge a camera to control audio")
            return
        player = self.players.get(self.enlarged)
        if not player:
            return
        self.enlarged_muted = not self.enlarged_muted
        player.set_muted(self.enlarged_muted)
        self._set_status("Muted" if self.enlarged_muted else "Audio on")

    def toggle_fullscreen(self) -> None:
        window = self.get_window()
        if window and window.get_state() & Gdk.WindowState.FULLSCREEN:
            self.unfullscreen()
        else:
            self.fullscreen()

    def _on_key(self, _widget, event) -> bool:
        key = event.keyval
        if key == Gdk.KEY_Escape:
            if self.enlarged:
                self._collapse()
            else:
                self.unfullscreen()
            return True
        if key in (Gdk.KEY_r, Gdk.KEY_R):
            self.reload()
            return True
        if key in (Gdk.KEY_s, Gdk.KEY_S):
            self.snapshot()
            return True
        if key in (Gdk.KEY_m, Gdk.KEY_M):
            self.toggle_mute()
            return True
        if key == Gdk.KEY_F11:
            self.toggle_fullscreen()
            return True
        return False
```

```python
from __future__ import annotations

import argparse
import shutil
import subprocess
import sys
from pathlib import Path

from .config import DEFAULT_CONFIG, load_cameras
from .util import redact_url


def check(config_path: Path) -> int:
    cameras = load_cameras(config_path)
    if not cameras:
        print(f"No cameras in {config_path}")
        return 1

    failed = 0
    for camera in cameras:
        try:
            result = subprocess.run(
                [
                    "ffprobe",
                    "-v",
                    "error",
                    "-rtsp_transport",
                    "tcp",
                    "-select_streams",
                    "v:0",
                    "-show_entries",
                    "stream=codec_name,width,height",
                    "-of",
                    "csv=p=0",
                    camera.url,
                ],
                capture_output=True,
                text=True,
                timeout=15,
            )
            ok = result.returncode == 0 and bool(result.stdout.strip())
            detail = result.stdout.strip() or result.stderr.strip()
        except (OSError, subprocess.TimeoutExpired) as exc:
            ok = False
            detail = str(exc)

        label = "OK  " if ok else "FAIL"
        print(f"{label} {camera.name} {redact_url(camera.url)}")
        if not ok and detail:
            print(f"     {detail.splitlines()[-1]}")
        failed += 0 if ok else 1

    return 0 if failed == 0 else 1


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(prog="wyze-view")
    parser.add_argument("--config", type=Path, default=DEFAULT_CONFIG)
    parser.add_argument(
        "--check", action="store_true", help="validate config and probe each camera"
    )
    args = parser.parse_args(argv)

    if args.check:
        return check(args.config)

    import gi

    gi.require_version("Gtk", "3.0")
    from gi.repository import Gtk

    if shutil.which("mpv") is None:
        dialog = Gtk.MessageDialog(
            message_type=Gtk.MessageType.ERROR,
            buttons=Gtk.ButtonsType.CLOSE,
            text="mpv is required",
        )
        dialog.format_secondary_text("Install it with: sudo apt install mpv")
        dialog.run()
        dialog.destroy()
        return 1

    from .app import MainWindow

    window = MainWindow(args.config)
    window.show_all()
    Gtk.main()
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

## Self-Review Notes

- Spec coverage: config/add-camera (Task 1), snapshot paths and redaction (Task 2), auto grid (Task 3), mpv embedding/reconnect/mute/snapshot (Task 4), click-to-enlarge, audio, fullscreen, offline status (Task 5), `--check`, launcher, menu entry (Task 6). Recording/PTZ/remote remain out of scope by design.
- The GUI has no automated tests because GTK embedding and mpv playback require a display and real cameras; all pure logic (config, paths, grid math, args, backoff, IPC framing, CLI checks) is covered by pytest, and the GUI steps are explicit manual checks.
- Names and signatures are consistent across tasks: `Camera(name, url, sub_url)`, `load_cameras`, `grid_dims` returns `(rows, cols)`, `Player(camera, xid, socket_path, on_state)`, `snapshot_path(directory, name, when)`, `redact_url`.
- Tiles live in a `Gtk.Fixed`, never reparented: GTK destroys a widget's GdkWindow (and its XID) on unparent, which would break mpv's `--wid`. Layout sets positions and sizes instead; hidden cameras are paused to save CPU.
- `GdkX11` is imported for its side effect so `Gdk.Window.get_xid()` is available on the X11 backend.
